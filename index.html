import Replicate from "replicate"; export default async function handler(req, res) { if (req.method !== 'POST') return res.status(405).json({error: 'POST only'}); const replicate = new Replicate({ auth: process.env.REPLICATE_API_TOKEN, }); const { prompt } = req.body; // ترجمة بسيطة let en = prompt; if(prompt.includes("عطبرة") || prompt.includes("ولد سوداني")) en = "Young Sudanese boy walking in streets of Atbara, cinematic photorealistic, late 90s indie film, golden hour, dusty street"; if(prompt.includes("حصان")) en = "beautiful white horse cinematic photorealistic"; if(prompt.includes("قطه")) en = "cute cat playing football cinematic"; try { const output = await replicate.run( "google/imagen-4", { input: { prompt: `Cinematic photorealistic medium shot, ${en}, warm nostalgic 90s film grain`, aspect_ratio: "16:9", safety_filter_level: "block_medium_and_above" } } ); res.status(200).json({ image: output }); } catch(e) { res.status(500).json({ error: e.message }); } }
